-- ===============================================================================================================
-- QUESTION 1
-- Original Author: Store Operations Manager
-- Description:     What are our top five best-selling products by total sales revenue this quarter, 
--                  and how do their current stock levels look?
-- ===============================================================================================================

-- Q1 Answer SQL Statement:
SELECT TOP 5
    p.product_id,
    p.product_name,
    p.category,
    SUM(s.quantity_sold * s.unit_price) AS total_revenue,
    SUM(s.quantity_sold) AS total_units_sold,
    p.current_stock_level
FROM 
    dbo.Products p
INNER JOIN dbo.SalesTransactions s ON p.product_id = s.product_id
WHERE 
    s.transaction_date >= '2026-07-01'
GROUP BY 
    p.product_id, p.product_name, p.category, p.current_stock_level
ORDER BY 
    total_revenue DESC;


-- ===============================================================================================================
-- QUESTION 2
-- Original Author: Financial Controller
-- Description:     What is the month-over-month profit margin growth rate for Ogesandy Venture 
--                  across our primary operating branches?
-- ===============================================================================================================

-- Q2 Answer SQL Statement:
WITH MonthlyFinancials AS (
    SELECT 
        branch_name,
        DATEPART(month, transaction_date) AS sales_month,
        DATEPART(year, transaction_date) AS sales_year,
        SUM(sales_amount) AS total_sales,
        SUM(operating_cost) AS total_cost,
        (SUM(sales_amount) - SUM(operating_cost)) AS net_profit
    FROM 
        dbo.BranchFinancials
    GROUP BY 
        branch_name, DATEPART(month, transaction_date), DATEPART(year, transaction_date)
)
SELECT 
    branch_name,
    sales_year,
    sales_month,
    total_sales,
    net_profit,
    CAST(net_profit AS FLOAT) / NULLIF(total_sales, 0) * 100 AS profit_margin_percentage,
    LAG(CAST(net_profit AS FLOAT) / NULLIF(total_sales, 0) * 100) OVER (PARTITION BY branch_name ORDER BY sales_year, sales_month) AS prev_month_margin
FROM 
    MonthlyFinancials;


-- ===============================================================================================================
-- QUESTION 3
-- Original Author: Customer Relations Lead
-- Description:     Which customer segments show the highest customer lifetime value (LTV) and 
--                  the lowest retention/churn risk?
-- ===============================================================================================================

-- Q3 Answer SQL Statement:
SELECT 
    customer_segment,
    COUNT(customer_id) AS total_customers,
    AVG(lifetime_spend) AS average_lifetime_value,
    AVG(days_since_last_purchase) AS avg_recency_days,
    SUM(CASE WHEN days_since_last_purchase > 90 THEN 1 ELSE 0 END) AS churned_customers,
    (CAST(SUM(CASE WHEN days_since_last_purchase > 90 THEN 1 ELSE 0 END) AS FLOAT) / COUNT(customer_id)) * 100 AS churn_rate_percentage
FROM 
    dbo.CustomerMetrics
GROUP BY 
    customer_segment
ORDER BY 
    average_lifetime_value DESC;


-- ===============================================================================================================
-- QUESTION 4 
-- Original Author: Joshua Asuquo
-- Description:     How does daily order volume correlate with weekday delivery performance and courier delay logs?
-- ===============================================================================================================

-- Q4 Answer SQL Statement:
SELECT 
    DATENAME(weekday, order_date) AS day_of_week,
    COUNT(order_id) AS total_daily_orders,
    AVG(actual_delivery_days) AS avg_delivery_time_days,
    SUM(CASE WHEN delivery_status = 'Delayed' THEN 1 ELSE 0 END) AS total_delayed_orders,
    CAST(SUM(CASE WHEN delivery_status = 'Delayed' THEN 1 ELSE 0 END) AS FLOAT) / COUNT(order_id) * 100 AS delay_rate_percentage
FROM 
    dbo.LogisticsPerformance
GROUP BY 
    DATENAME(weekday, order_date), DATEPART(weekday, order_date)
ORDER BY 
    DATEPART(weekday, order_date);
